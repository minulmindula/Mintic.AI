# Page 1: Core Risk Model & Unbendable Safety Lock Engine

The foundational principle of Mintic.AI is that **no AI model has authority to execute trades directly**. All trade candidates generated by generative agents must pass through the deterministic, zero-AI `UnbendableSafetyLock` engine.

### 1.1 Strict Pydantic Trade Proposal & Safety Result Models
**File**: [`apps/api/src/core/schemas.py`](file:///Users/minulmindula/Development/Mintic.AI/SOURCE/Mintic.AI/apps/api/src/core/schemas.py)

```python
class TradeProposal(BaseModel):
    """Pydantic model representing a trade proposal submitted for safety verification."""
    model_config = ConfigDict(frozen=True, extra="forbid")

    symbol: str = Field(..., min_length=1, description="Ticker symbol (e.g. AAPL)")
    action: Literal["BUY", "SELL"] = Field(..., description="Action: BUY or SELL")
    quantity: float = Field(..., gt=0, description="Asset units to trade (> 0)")
    confidence_score: float = Field(..., ge=0.0, le=1.0, description="Score 0.0 to 1.0")
    target_price: float = Field(..., gt=0, description="Target price per unit in GBP (£)")
    stop_loss: float = Field(..., gt=0, description="Stop loss price in GBP (£)")
    reasoning_summary: str = Field(..., min_length=1, description="Trade rationale summary")


class SafetyCheckResult(BaseModel):
    """Result of evaluating an UnbendableSafetyLock check."""
    model_config = ConfigDict(frozen=True, extra="forbid")

    is_approved: bool = Field(..., description="Safety lock approval status")
    reason: str = Field(..., description="Human-readable decision explanation")
    proposal: TradeProposal = Field(..., description="Evaluated trade proposal")
    failed_checks: list[str] = Field(default_factory=list, description="Violated rules")
```

### 1.3 AI-Free Safety Lock Evaluation Logic
**File**: [`apps/api/src/core/safety.py`](file:///Users/minulmindula/Development/Mintic.AI/SOURCE/Mintic.AI/apps/api/src/core/safety.py)

```python
class UnbendableSafetyLock:
    """Deterministic Safety Lock - Zero AI / LLM dependencies allowed."""

    def validate_trade(
        self, proposal: TradeProposal, current_drawdown_percent: float = 0.0
    ) -> SafetyCheckResult:
        failed_checks: list[str] = []

        # 1. Stock Whitelist Check
        if proposal.symbol.upper() not in self.settings.ALLOWED_SYMBOLS:
            failed_checks.append(
                f"Symbol '{proposal.symbol}' is not in allowed stock whitelist"
            )

        # 2. Position Size Limit Check in GBP (£)
        position_value_gbp = proposal.quantity * proposal.target_price
        if position_value_gbp > self.settings.MAX_POSITION_SIZE_GBP:
            failed_checks.append(
                f"Position size £{position_value_gbp:.2f} exceeds max £{self.settings.MAX_POSITION_SIZE_GBP:.2f}"
            )

        # 3. Minimum Confidence Score Check
        if proposal.confidence_score < self.settings.MIN_CONFIDENCE_THRESHOLD:
            failed_checks.append(
                f"Confidence score {proposal.confidence_score:.2f} below min threshold {self.settings.MIN_CONFIDENCE_THRESHOLD:.2f}"
            )

        # 4. Daily Drawdown Threshold Check
        if current_drawdown_percent > self.settings.MAX_DAILY_DRAWDOWN_PERCENT:
            failed_checks.append(
                f"Current drawdown {current_drawdown_percent:.2f}% exceeds max {self.settings.MAX_DAILY_DRAWDOWN_PERCENT:.2f}%"
            )

        is_approved = len(failed_checks) == 0
        reason = "All safety checks passed successfully." if is_approved else f"Safety lock rejected: {'; '.join(failed_checks)}"

        return SafetyCheckResult(
            is_approved=is_approved,
            reason=reason,
            proposal=proposal,
            failed_checks=failed_checks,
        )
```

```python
def get_default_llm() -> ChatGoogleGenerativeAI:
    """Instantiate ChatGoogleGenerativeAI with gemini-1.5-flash."""
    api_key = settings.GEMINI_API_KEY or settings.GOOGLE_API_KEY or os.getenv("GEMINI_API_KEY")
    return ChatGoogleGenerativeAI(model="gemini-1.5-flash", temperature=0.2, google_api_key=api_key)


def risk_manager_node(state: DeliberationState, llm: Optional[Any] = None) -> dict[str, Any]:
    """Chief Risk Officer Node: Enforces structured Pydantic output."""
    model = llm or get_default_llm()
    structured_llm = model.with_structured_output(TradeProposal)

    messages = [
        SystemMessage(content="Review analyst report, bull hypothesis, and bear hypothesis to issue a final TradeProposal JSON."),
        HumanMessage(content=f"Symbol: {state.get('symbol')}\nAnalyst: {state.get('analyst_report')}\nBull: {state.get('bull_hypothesis')}\nBear: {state.get('bear_hypothesis')}"),
    ]
    proposal: TradeProposal = structured_llm.invoke(messages)
    return {"final_proposal": proposal}
```

### 2.3 StateGraph Compilation
**File**: [`apps/api/src/agents/graph.py`](file:///Users/minulmindula/Development/Mintic.AI/SOURCE/Mintic.AI/apps/api/src/agents/graph.py)

```python
def create_deliberation_graph() -> CompiledStateGraph:
    """Compiles LangGraph execution workflow."""
    workflow = StateGraph(DeliberationState)

    workflow.add_node("analyst", analyst_node)
    workflow.add_node("bull", bull_node)
    workflow.add_node("bear", bear_node)
    workflow.add_node("risk_manager", risk_manager_node)

    workflow.add_edge(START, "analyst")
    workflow.add_edge("analyst", "bull")
    workflow.add_edge("analyst", "bear")
    workflow.add_edge("bull", "risk_manager")
    workflow.add_edge("bear", "risk_manager")
    workflow.add_edge("risk_manager", END)

    return workflow.compile()
```

### 2.4 Sandboxed Execution Enforcement
**File**: [`apps/api/src/execution/sandbox_broker.py`](file:///Users/minulmindula/Development/Mintic.AI/SOURCE/Mintic.AI/apps/api/src/execution/sandbox_broker.py)

```python
class SandboxedBrokerService:
    """Executes trades against Alpaca paper API ONLY if is_approved is True."""

    def execute_trade(self, safety_result: SafetyCheckResult) -> Order:
        if not safety_result.is_approved:
            raise PermissionError(
                f"EXECUTION DENIED: Safety lock rejected trade for '{safety_result.proposal.symbol}'. Reason: {safety_result.reason}"
            )

        # Enforce Alpaca Paper Trading Client Only
        order_data = MarketOrderRequest(
            symbol=safety_result.proposal.symbol.upper(),
            qty=safety_result.proposal.quantity,
            side=OrderSide.BUY if safety_result.proposal.action == "BUY" else OrderSide.SELL,
            time_in_force=TimeInForce.GTC,
        )
        return self.client.submit_order(order_data)
```

```python
@app.post("/api/v1/trigger-analysis", response_model=TriggerAnalysisResponse)
async def trigger_analysis(
    request: TriggerAnalysisRequest, background_tasks: BackgroundTasks
) -> TriggerAnalysisResponse:
    # 1. Multi-Agent Deliberation Graph Execution
    final_state: dict[str, Any] = {}
    try:
        final_state = await deliberation_workflow.ainvoke(initial_state)
        proposal = final_state.get("final_proposal")
    except Exception as err:
        proposal = TradeProposal(symbol=request.symbol, action="BUY", quantity=10.0, confidence_score=0.85, target_price=180.0, stop_loss=170.0, reasoning_summary=f"Fallback proposal ({err})")

    # 2. Deterministic Safety Lock Evaluation
    safety_eval: SafetyCheckResult = safety_lock.validate_trade(proposal=proposal)

    # 3. Sandboxed Broker Execution (Only if Approved)
    execution_output: Optional[dict[str, Any]] = None
    if safety_eval.is_approved:
        broker = SandboxedBrokerService()
        order_res = broker.execute_trade(safety_result=safety_eval)
        execution_output = {"order_id": order_res.id, "status": "submitted"}

    # 4. Background Audit Logging
    background_tasks.add_task(
        log_audit_entry,
        symbol=request.symbol,
        confidence_score=proposal.confidence_score,
        is_approved=safety_eval.is_approved,
        reasoning_summary=safety_eval.reason,
    )

    return TriggerAnalysisResponse(
        status="completed", symbol=request.symbol, is_approved=safety_eval.is_approved,
        proposal=proposal, safety_result={"is_approved": safety_eval.is_approved, "reason": safety_eval.reason, "failed_checks": safety_eval.failed_checks},
        execution_result=execution_output, analyst_report=final_state.get("analyst_report"),
        bull_hypothesis=final_state.get("bull_hypothesis"), bear_hypothesis=final_state.get("bear_hypothesis")
    )
```

### 3.2 Async SQLAlchemy Audit Log Engine

```python
class AuditLog(Base):
    __tablename__ = "audit_logs"
    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    symbol: Mapped[str] = mapped_column(String(32), nullable=False, index=True)
    confidence_score: Mapped[float] = mapped_column(Float, nullable=False)
    is_approved: Mapped[bool] = mapped_column(Boolean, nullable=False)
    reasoning_summary: Mapped[str] = mapped_column(Text, nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=lambda: datetime.now(timezone.utc))


async def log_audit_entry(symbol: str, confidence_score: float, is_approved: bool, reasoning_summary: str) -> AuditLog:
    session_factory = get_session_maker(get_engine())
    audit_entry = AuditLog(symbol=symbol, confidence_score=confidence_score, is_approved=is_approved, reasoning_summary=reasoning_summary)
    async with session_factory() as session:
        async with session.begin():
            session.add(audit_entry)
    return audit_entry
```
